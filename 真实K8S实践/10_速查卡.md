## 五、常用命令

```bash
# 集群
export KUBECONFIG=/etc/kubernetes/admin.conf
kubectl get nodes -o wide
kubectl -n kube-system get ds | grep -E 'cilium|kube-proxy'
cilium status                    # 需要 cilium-cli，1.20 用 cilium-dbg 更准

# Cilium 细节
kubectl -n kube-system exec ds/cilium -c cilium-agent -- cilium-dbg status | head -20
kubectl -n kube-system exec ds/cilium -c cilium-agent -- cilium-dbg endpoint list
kubectl -n kube-system exec ds/cilium -c cilium-agent -- cilium-dbg monitor --type drop

# Hubble（需 hubble CLI + 端口转发）
kubectl -n kube-system port-forward svc/hubble-relay 4245:80 &
hubble observe --server localhost:4245 --namespace apps --verdict DROPPED

# etcd
kubectl -n kube-system exec etcd-cp-1 -- etcdctl member list --cacert /etc/kubernetes/pki/etcd/ca.crt \
  --cert /etc/kubernetes/pki/etcd/server.crt --key /etc/kubernetes/pki/etcd/server.key
/usr/local/bin/etcd-snapshot.sh          # 手动快照

# 存储
kubectl -n longhorn-system get volumes.longhorn.io -o custom-columns=NAME:.metadata.name,ROBUST:.status.robustness
kubectl -n longhorn-system get backuptargets.longhorn.io default -o jsonpath='{.status.available}'

# 备份
velero backup get
velero backup create <name> --include-namespaces apps --default-volumes-to-fs-backup --wait
velero restore create --from-backup <name> --wait

# 交付
kubectl -n argocd get application apps -o jsonpath='{.status.sync.status} {.status.health.status}'
kubectl -n apps delete deploy podinfo --wait=false   # 演练自愈

# 巡检
bash /usr/local/bin/k8s-patrol.sh
bash /usr/local/bin/k8s-patrol.sh --check            # 有问题返回非 0
```
